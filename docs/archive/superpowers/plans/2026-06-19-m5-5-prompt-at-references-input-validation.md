# M5.5 Prompt @ References And Input Validation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let Studio users use Prompt `@` references to explicitly control node inputs, while making implicit and invalid inputs visible and preventing invalid runs.

**Architecture:** Keep the first version intentionally lightweight: textarea remains the prompt source, `@` opens a simple candidate menu, and selected refs are persisted in `media_node.prompt_refs` plus lightweight `prompt_rich`. Add shared pure helpers on both backend and frontend so explicit, implicit, and invalid input state is computed consistently. Keep dependency edges as the actual graph contract; `@` refs annotate and can auto-create missing dependency edges.

**Tech Stack:** Go 1.26, Hertz, pgx/sqlc, React 19, TypeScript 6, TanStack Query, React Flow, Vite 8, Node test runner, in-app browser smoke.

## Scope Boundaries

M5.5 includes:

- Stable prompt ref JSON node shared by frontend and backend.
- Node update API support for `prompt_refs` and `prompt_rich`.
- Prompt `@` menu in the inline node editor and property panel prompt editor.
- Candidate ordering: connected upstream nodes first, then other non-target workspace nodes.
- Selecting an unconnected candidate auto-creates a dependency edge.
- Explicit, implicit, and invalid input state helpers and UI display.
- Run button disabled reason for invalid refs.
- Backend run rejection for invalid refs even if the frontend is bypassed.
- Intent input refs distinguish ordinary direct inputs as `explicit` or `implicit`, while preserving `reference_pack` and `reference_pack_member`.
- E2E smoke for explicit ref, implicit ref, invalid ref, and auto-created dependency.

M5.5 does not include:

- Full rich text editor integration.
- Drag/drop or visual chip editing inside the textarea.
- Removing dependency edges when a prompt ref is removed.
- Multi-cursor prompt editing.
- Ref aliases beyond the selected node title snapshot.
- Agent mode prompt behavior.

## Data Contracts

Use this first-version `prompt_refs` JSON node:

```json
{
  "version": 1,
  "refs": [
    {
      "node_id": "018f6ef0-5f4f-7e86-9f8c-100000000001",
      "label": "主图 A",
      "node_type": "image"
    }
  ]
}
```

Use this lightweight `prompt_rich` JSON node:

```json
{
  "version": 1,
  "source": "textarea-at",
  "text": "Use @主图 A as style reference"
}
```

Behavioral rules:

- `prompt_template` and legacy `prompt` remain the rendered prompt text.
- `prompt_refs.refs[].node_id` is the explicit reference fact source.
- `prompt_refs.refs[].label` is a display snapshot; validation resolves the current node by `node_id`.
- Duplicate refs are normalized to the first occurrence.
- Empty prompt refs are stored as `{"version":1,"refs":[]}`.
- Invalid ref means the referenced node does not exist in the same workspace or no dependency edge exists from ref node to target node.

## Deliverable Standards

- Prompt refs persist through refresh as `prompt_refs`.
- Inline node editor shows an `@` candidate menu when the user types `@`.
- Candidate menu lists connected upstream inputs first.
- Candidate menu includes unconnected nodes except the target node itself.
- Selecting an unconnected node automatically creates a dependency edge.
- Removing a prompt ref updates `prompt_refs` only; it does not remove dependency edges.
- The UI shows explicit refs, implicit refs, and invalid refs separately.
- Invalid refs disable the run button and show a clear reason.
- Backend run rejects invalid refs even when called directly.
- Intent direct inputs use `kind = explicit` or `kind = implicit`.
- Intent Reference Pack inputs keep `kind = reference_pack` and expanded members keep `kind = reference_pack_member`.

## Acceptance Standards

- A -> B dependency exists; B prompt has `@A`; running B produces latest job intent with A as `explicit`.
- A -> B dependency exists; B prompt does not have `@A`; UI shows A as `implicit`, and running B produces latest job intent with A as `implicit`.
- B prompt has `@A`; deleting A -> B dependency makes B show invalid ref and prevents successful run.
- B prompt uses `@C` when C is not connected; the system creates C -> B dependency and writes C into `prompt_refs`.
- Refreshing the page preserves prompt text, prompt refs, and explicit/implicit/invalid display.

```go
package promptrefs

import (
	"encoding/json"
	"fmt"
	"strings"

	"github.com/jackc/pgx/v5/pgtype"
)

type Document struct {
	Version int   `json:"version"`
	Refs    []Ref `json:"refs"`
}

type Ref struct {
	NodeID   string `json:"node_id"`
	Label    string `json:"label"`
	NodeType string `json:"node_type"`
}

type RichDocument struct {
	Version int    `json:"version"`
	Source  string `json:"source"`
	Text    string `json:"text"`
}

func Normalize(raw []byte) (Document, []byte, error) {
	if len(raw) == 0 || strings.TrimSpace(string(raw)) == "" || strings.TrimSpace(string(raw)) == "null" {
		return emptyDocument()
	}
	var doc Document
	if err := json.Unmarshal(raw, &doc); err != nil {
		return Document{}, nil, err
	}
	if doc.Version == 0 {
		doc.Version = 1
	}
	if doc.Version != 1 {
		return Document{}, nil, fmt.Errorf("unsupported prompt refs version %d", doc.Version)
	}
	seen := map[string]bool{}
	refs := make([]Ref, 0, len(doc.Refs))
	for _, ref := range doc.Refs {
		ref.NodeID = strings.TrimSpace(ref.NodeID)
		ref.Label = strings.TrimSpace(ref.Label)
		ref.NodeType = strings.TrimSpace(ref.NodeType)
		if ref.NodeID == "" {
			return Document{}, nil, fmt.Errorf("prompt ref missing node_id")
		}
		if _, ok := parseUUID(ref.NodeID); !ok {
			return Document{}, nil, fmt.Errorf("invalid prompt ref node_id")
		}
		if seen[ref.NodeID] {
			continue
		}
		seen[ref.NodeID] = true
		refs = append(refs, ref)
	}
	doc.Refs = refs
	normalized, err := json.Marshal(doc)
	return doc, normalized, err
}

func NormalizeRich(raw []byte, prompt string) ([]byte, error) {
	if len(raw) == 0 || strings.TrimSpace(string(raw)) == "" || strings.TrimSpace(string(raw)) == "null" {
		return json.Marshal(RichDocument{Version: 1, Source: "textarea-at", Text: prompt})
	}
	var doc RichDocument
	if err := json.Unmarshal(raw, &doc); err != nil {
		return nil, err
	}
	if doc.Version == 0 {
		doc.Version = 1
	}
	if doc.Version != 1 {
		return nil, fmt.Errorf("unsupported prompt rich version %d", doc.Version)
	}
	if strings.TrimSpace(doc.Source) == "" {
		doc.Source = "textarea-at"
	}
	if doc.Text == "" {
		doc.Text = prompt
	}
	return json.Marshal(doc)
}

func emptyDocument() (Document, []byte, error) {
	doc := Document{Version: 1, Refs: []Ref{}}
	normalized, err := json.Marshal(doc)
	return doc, normalized, err
}

func parseUUID(value string) (pgtype.UUID, bool) {
	var id pgtype.UUID
	if err := id.Scan(value); err != nil {
		return pgtype.UUID{}, false
	}
	return id, id.Valid
}
```

```go
func invalidPromptRefs(target db.MediaNode, upstream []db.MediaNode) ([]promptrefs.Ref, error) {
	doc, _, err := promptrefs.Normalize(target.PromptRefs)
	if err != nil {
		return nil, err
	}
	upstreamIDs := map[string]bool{}
	for _, dep := range upstream {
		upstreamIDs[uuidToString(dep.ID)] = true
	}
	invalid := []promptrefs.Ref{}
	for _, ref := range doc.Refs {
		if !upstreamIDs[ref.NodeID] {
			invalid = append(invalid, ref)
		}
	}
	return invalid, nil
}
```

```ts
import type { MediaEdge, MediaNode, PromptRef, PromptRefsDocument } from "./api";

export interface InputReferenceState {
  explicit: Array<{ node: MediaNode; ref: PromptRef }>;
  implicit: MediaNode[];
  invalid: Array<{ ref: PromptRef }>;
}

export function normalizePromptRefs(value: MediaNode["prompt_refs"]): PromptRefsDocument {
  if (!value || typeof value !== "object" || !Array.isArray(value.refs)) {
    return { version: 1, refs: [] };
  }
  const seen = new Set<string>();
  const refs: PromptRef[] = [];
  for (const ref of value.refs) {
    if (!ref?.node_id || seen.has(ref.node_id)) {
      continue;
    }
    seen.add(ref.node_id);
    refs.push({
      node_id: ref.node_id,
      label: ref.label || "未命名",
      node_type: ref.node_type,
    });
  }
  return { version: 1, refs };
}

export function upstreamNodesForTarget(target: Pick<MediaNode, "id">, nodes: MediaNode[], edges: MediaEdge[]) {
  const upstreamIDs = edges.filter((edge) => edge.to_node_id === target.id).map((edge) => edge.from_node_id);
  const byID = new Map(nodes.map((node) => [node.id, node]));
  return upstreamIDs.map((id) => byID.get(id)).filter((node): node is MediaNode => Boolean(node));
}

export function candidatePromptRefNodes(target: Pick<MediaNode, "id">, nodes: MediaNode[], edges: MediaEdge[]) {
  const upstream = upstreamNodesForTarget(target, nodes, edges);
  const upstreamIDs = new Set(upstream.map((node) => node.id));
  const rest = nodes.filter((node) => node.id !== target.id && !upstreamIDs.has(node.id));
  return [...upstream, ...rest];
}

export function promptRefsAfterSelect(current: PromptRefsDocument, node: Pick<MediaNode, "id" | "node_type" | "title">) {
  const refs = normalizePromptRefs(current).refs;
  if (refs.some((ref) => ref.node_id === node.id)) {
    return { version: 1 as const, refs };
  }
  return {
    version: 1 as const,
    refs: [...refs, { node_id: node.id, label: node.title || "未命名", node_type: node.node_type }],
  };
}

export function promptRefsAfterRemove(current: PromptRefsDocument, nodeId: string) {
  return {
    version: 1 as const,
    refs: normalizePromptRefs(current).refs.filter((ref) => ref.node_id !== nodeId),
  };
}

export function inputReferenceState(target: MediaNode, nodes: MediaNode[], edges: MediaEdge[]): InputReferenceState {
  const refs = normalizePromptRefs(target.prompt_refs).refs;
  const nodesByID = new Map(nodes.map((node) => [node.id, node]));
  const upstream = upstreamNodesForTarget(target, nodes, edges);
  const upstreamIDs = new Set(upstream.map((node) => node.id));
  const explicit: InputReferenceState["explicit"] = [];
  const invalid: InputReferenceState["invalid"] = [];
  const explicitIDs = new Set<string>();
  for (const ref of refs) {
    const node = nodesByID.get(ref.node_id);
    if (!node || !upstreamIDs.has(ref.node_id)) {
      invalid.push({ ref });
      continue;
    }
    explicitIDs.add(ref.node_id);
    explicit.push({ node, ref });
  }
  const implicit = upstream.filter((node) => !explicitIDs.has(node.id));
  return { explicit, implicit, invalid };
}

export function runDisabledReasonForPromptRefs(state: Pick<InputReferenceState, "invalid">) {
  return state.invalid.length > 0
    ? "Prompt 中有失效引用，请重新选择或移除后再运行。"
    : null;
}
```
